# Presentation-rule tests (dark remake 1C, implementation-plan.md 4 / 8).
#
# The UE module (Source/Abyssfire) keeps the decisions of its presentation glue in UE-free, header-only helpers
# (Source/Abyssfire/Private/<Area>/Abyss*Rules.h: std + AbyssCore public headers only, no CoreMinimal, no UE macros), so
# they build and run here without the engine. This target links the portable core (same block and flags as
# CoreTests/CMakeLists.txt: C++20, no exceptions, no RTTI, no FP contraction, -Werror) and one doctest runner over
# tests/*.cpp. Tests live here, never under Source/Abyssfire (UBT compiles every .cpp in that tree).
#
# Adding a chain's tests: drop tests/<area>_test.cpp (doctest TEST_SUITE "<area>"), no CMake edit; the single ctest
# entry `present` runs every suite. shim/PureMacroShim.cpp compiles every Abyss*Rules.h after the UE / Windows / Apple
# macros a game-module TU sees (the generated present_shim_rules.h lists them), so a helper that names check(), min(),
# TEXT, ... fails here instead of in the Mac build.
cmake_minimum_required(VERSION 3.20)
project(AbyssPresentationTests LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE RelWithDebInfo CACHE STRING "" FORCE)
endif()

option(ABYSS_SANITIZE "Build with AddressSanitizer + UndefinedBehaviorSanitizer" OFF)

get_filename_component(ABYSS_UNREAL_DIR "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
set(ABYSS_CORE_DIR "${ABYSS_UNREAL_DIR}/Source/AbyssCore")
set(ABYSS_MODULE_PRIVATE_DIR "${ABYSS_UNREAL_DIR}/Source/Abyssfire/Private")

if(MSVC)
  set(ABYSS_CORE_FLAGS /W4 /WX /we4668 /wd4100 /bigobj /GR- /EHs-c- /D_HAS_EXCEPTIONS=0 /fp:precise /utf-8)
else()
  set(ABYSS_CORE_FLAGS -Wall -Wextra -Wshadow -Wundef -Werror -Wno-unused-parameter -fno-exceptions -fno-rtti
      -ffp-contract=off)
endif()

if(ABYSS_SANITIZE AND NOT MSVC)
  add_compile_options(-fsanitize=address,undefined,float-cast-overflow -fno-omit-frame-pointer
                      -fno-sanitize-recover=undefined,float-cast-overflow)
  add_link_options(-fsanitize=address,undefined,float-cast-overflow)
endif()

# ---- the core library (as CoreTests builds it) -----------------------------------------------------------------------
file(GLOB_RECURSE ABYSS_CORE_SOURCES CONFIGURE_DEPENDS "${ABYSS_CORE_DIR}/Private/*.cpp")
list(FILTER ABYSS_CORE_SOURCES EXCLUDE REGEX ".*/Private/UE/.*")
file(GLOB_RECURSE ABYSS_CORE_HEADERS CONFIGURE_DEPENDS "${ABYSS_CORE_DIR}/Public/*.h" "${ABYSS_CORE_DIR}/Private/*.h")
add_library(AbyssCore STATIC ${ABYSS_CORE_SOURCES} ${ABYSS_CORE_HEADERS})
target_include_directories(AbyssCore PUBLIC "${ABYSS_CORE_DIR}/Public")
target_compile_definitions(AbyssCore PRIVATE ABYSS_CORE_BUILDING=1)
target_compile_options(AbyssCore PRIVATE ${ABYSS_CORE_FLAGS})

# ---- the pure presentation helpers -----------------------------------------------------------------------------------
file(GLOB_RECURSE ABYSS_RULE_HEADERS CONFIGURE_DEPENDS RELATIVE "${ABYSS_MODULE_PRIVATE_DIR}"
  "${ABYSS_MODULE_PRIVATE_DIR}/*Rules.h")
list(FILTER ABYSS_RULE_HEADERS INCLUDE REGEX "(^|/)Abyss[A-Za-z0-9]*Rules\\.h$")
list(SORT ABYSS_RULE_HEADERS)
set(ABYSS_SHIM_RULES "")
set(ABYSS_SHIM_STD "")
foreach(h IN LISTS ABYSS_RULE_HEADERS)
  file(STRINGS "${ABYSS_MODULE_PRIVATE_DIR}/${h}" std_lines REGEX "^#include <[^>]+>")
  foreach(line IN LISTS std_lines)
    string(REGEX MATCH "<[^>]+>" inc "${line}")
    list(APPEND ABYSS_SHIM_STD "${inc}")
  endforeach()
  string(APPEND ABYSS_SHIM_RULES "#include \"${h}\"\n")
endforeach()
# The core public headers the helpers include bring their own <...> headers: those come before the macros too.
file(GLOB_RECURSE ABYSS_PUBLIC_HEADERS CONFIGURE_DEPENDS "${ABYSS_CORE_DIR}/Public/*.h")
foreach(h IN LISTS ABYSS_PUBLIC_HEADERS)
  file(STRINGS "${h}" std_lines REGEX "^#include <[^>]+>")
  foreach(line IN LISTS std_lines)
    string(REGEX MATCH "<[^>]+>" inc "${line}")
    list(APPEND ABYSS_SHIM_STD "${inc}")
  endforeach()
endforeach()
list(REMOVE_DUPLICATES ABYSS_SHIM_STD)
list(SORT ABYSS_SHIM_STD)
set(ABYSS_SHIM_STD_TEXT "")
foreach(inc IN LISTS ABYSS_SHIM_STD)
  string(APPEND ABYSS_SHIM_STD_TEXT "#include ${inc}\n")
endforeach()
set(ABYSS_SHIM_GEN "${CMAKE_CURRENT_BINARY_DIR}/shim")
file(CONFIGURE OUTPUT "${ABYSS_SHIM_GEN}/present_shim_std.h" CONTENT "// generated by CMakeLists.txt\n${ABYSS_SHIM_STD_TEXT}")
file(CONFIGURE OUTPUT "${ABYSS_SHIM_GEN}/present_shim_rules.h" CONTENT "// generated by CMakeLists.txt\n${ABYSS_SHIM_RULES}")

# ---- tests ------------------------------------------------------------------------------------------------------------
file(GLOB ABYSS_TEST_SOURCES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/tests/*.cpp")
add_executable(present_tests ${ABYSS_TEST_SOURCES})
target_link_libraries(present_tests PRIVATE AbyssCore)
target_include_directories(present_tests SYSTEM PRIVATE "${ABYSS_UNREAL_DIR}/CoreTests/third_party")
target_include_directories(present_tests PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/tests" "${ABYSS_MODULE_PRIVATE_DIR}")
target_compile_definitions(present_tests PRIVATE
  DOCTEST_CONFIG_NO_EXCEPTIONS
  DOCTEST_CONFIG_NO_EXCEPTIONS_BUT_WITH_ALL_ASSERTS
  DOCTEST_CONFIG_SUPER_FAST_ASSERTS
  ABYSS_UNREAL_DIR="${ABYSS_UNREAL_DIR}")
target_compile_options(present_tests PRIVATE ${ABYSS_CORE_FLAGS})

# ---- macro shim (compile-only) ----------------------------------------------------------------------------------------
add_library(present_macro_shim OBJECT "${CMAKE_CURRENT_SOURCE_DIR}/shim/PureMacroShim.cpp")
target_include_directories(present_macro_shim PRIVATE "${ABYSS_CORE_DIR}/Public" "${ABYSS_MODULE_PRIVATE_DIR}" "${ABYSS_SHIM_GEN}")
target_compile_definitions(present_macro_shim PRIVATE ABYSS_CORE_DLL=1)  # consumer side of a modular build, like the module
target_compile_options(present_macro_shim PRIVATE ${ABYSS_CORE_FLAGS})
add_dependencies(present_tests present_macro_shim)

enable_testing()
add_test(NAME present COMMAND present_tests --no-intro --no-version)
